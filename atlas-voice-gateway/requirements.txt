# Atlas Voice Gateway - Python dependencies
#
# Install into a virtual environment on HAL:
#   python3 -m venv /opt/atlas-voice-gateway/.venv
#   /opt/atlas-voice-gateway/.venv/bin/pip install -r requirements.txt
#
# GPU note: faster-whisper needs a CUDA-capable PyTorch runtime for GPU use.
# On an RTX A2000 the usual path is to install a CUDA build of torch, e.g.:
#   pip install torch --index-url https://download.pytorch.org/whl/cu121
# (faster-whisper pulls a default torch build; align it with the CUDA version
#  reported by `nvidia-smi` on HAL. Verify on the machine.)
#
# ffmpeg is a SYSTEM dependency (not pip) and is required for non-WAV audio
# (browser WebM/Opus). Install it with the OS package manager, e.g.:
#   sudo apt-get install -y ffmpeg

fastapi>=0.115.0
uvicorn[standard]>=0.30.0
httpx>=0.27.0

# Multipart form uploads (WAV/WebM audio) for /api/transcribe and /api/voice.
python-multipart>=0.0.9

# Local speech-to-text on HAL.
faster-whisper>=1.0.3

# Local text-to-speech (Piper). The Python package pulls the ONNX runtime used
# for CPU inference, which is exactly what we want: TTS stays on the CPU and
# does not compete with faster-whisper / Ollama for the RTX A2000.
# Voice model files (.onnx + .onnx.json) are downloaded separately - see README.
piper-tts>=1.2.0
